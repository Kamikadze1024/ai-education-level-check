1) # аутентификация пользователей
POST /core/auth

тело запроса - json вида:
{"login":"<login_value>", "password": "<password_value>"}

ответ:
{"result": "ok/fail"}


2) # генерация списка вопросов
POST /gen/questions

тело запроса - json вида:
{
    "num_questions": x, //к-во запросов
    "num_answ_per_one_quest": y, //к-во вариантов ответа на один вопрос
    "num_correct_answ_per_one_quest": z //к-во правильных ответов на
}

ответ - json вида:
{
    "msg_type": "questions_list", //это просто название поля (постоянно)
    "questions": [
        {
            "question_num": 1, //номер вопроса
            "question_txt": "<формулировка вопроса>", //текстовка вопроса
            "correct_answs": [//массив правильных ответов
                {
                    "answ_txt": "<текст правильного ответа>"
                }
            ],
            "not_correct_answs": [//массив неправильных ответов
                {
                    "answ_txt": "<текст неправильного ответа>"
                }
            ]
        }
    ]
}
